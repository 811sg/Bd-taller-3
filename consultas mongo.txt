use tienda_online

// 1. Mejores ventas: Top 10 productos más vendidos
db.pedidos.aggregate([
  { $match: { estado: { $ne: "Cancelado" } } },
  { $unwind: "$items" },
  {
    $group: {
      _id: "$items.producto_id",
      unidades: { $sum: "$items.cantidad" },
      ingresos: { $sum: "$items.subtotal" }
    }
  },
  { $sort: { unidades: -1 } },
  { $limit: 10 },
  {
    $lookup: {
      from: "productos",
      localField: "_id",
      foreignField: "_id",
      as: "producto"
    }
  },
  { $unwind: "$producto" },
  {
    $project: {
      _id: 0,
      producto: "$producto.nombre",
      categoria: "$producto.categoria.nombre",
      stock_actual: "$producto.stock",
      unidades: 1,
      ingresos: 1
    }
  }
])

// 2. Ingresos por categoría
db.pedidos.aggregate([
  { $match: { estado: { $ne: "Cancelado" } } },
  { $unwind: "$items" },
  {
    $lookup: {
      from: "productos",
      localField: "items.producto_id",
      foreignField: "_id",
      as: "producto"
    }
  },
  { $unwind: "$producto" },
  {
    $group: {
      _id: "$producto.categoria.nombre",
      ingresos: { $sum: "$items.subtotal" },
      unidades: { $sum: "$items.cantidad" }
    }
  },
  { $sort: { ingresos: -1 } },
  {
    $project: {
      _id: 0,
      categoria: "$_id",
      ingresos: 1,
      unidades: 1
    }
  }
])

// 3. Mejores 10 clientes por gasto (pedidos entregados)
db.pedidos.aggregate([
  { $match: { estado: "Entregado" } },
  {
    $group: {
      _id: "$cliente_id",
      gasto_total: { $sum: "$total" },
      pedidos: { $sum: 1 }
    }
  },
  { $sort: { gasto_total: -1 } },
  { $limit: 10 },
  {
    $lookup: {
      from: "clientes",
      localField: "_id",
      foreignField: "_id",
      as: "cliente"
    }
  },
  { $unwind: "$cliente" },
  {
    $project: {
      _id: 0,
      cliente: "$cliente.nombre",
      email: "$cliente.email",
      pedidos: 1,
      gasto_total: 1,
      ticket_promedio: {
        $round: [
          { $divide: ["$gasto_total", "$pedidos"] },
          0
        ]
      }
    }
  }
])

// 4. Ventas mensuales
db.pedidos.aggregate([
  { $match: { estado: { $ne: "Cancelado" } } },
  {
    $group: {
      _id: {
        anio: { $year: "$fecha" },
        mes: { $month: "$fecha" }
      },
      ventas: { $sum: "$total" },
      pedidos: { $sum: 1 }
    }
  },
  { $sort: { "_id.anio": 1, "_id.mes": 1 } },
  {
    $project: {
      _id: 0,
      anio: "$_id.anio",
      mes: "$_id.mes",
      ventas: 1,
      pedidos: 1
    }
  }
])

// 5. Productos mejor calificados con su proveedor (múltiples $lookup)
db.resenas.aggregate([
  {
    $group: {
      _id: "$producto_id",
      promedio: { $avg: "$calificacion" },
      num_resenas: { $sum: 1 }
    }
  },
  { $match: { num_resenas: { $gte: 3 } } },
  { $sort: { promedio: -1, num_resenas: -1 } },
  { $limit: 10 },
  {
    $lookup: {
      from: "productos",
      localField: "_id",
      foreignField: "_id",
      as: "producto"
    }
  },
  { $unwind: "$producto" },
  {
    $lookup: {
      from: "proveedores",
      localField: "producto.proveedor_id",
      foreignField: "_id",
      as: "proveedor"
    }
  },
  { $unwind: "$proveedor" },
  {
    $project: {
      _id: 0,
      producto: "$producto.nombre",
      categoria: "$producto.categoria.nombre",
      proveedor: "$proveedor.nombre",
      ciudad_proveedor: "$proveedor.ciudad",
      promedio: { $round: ["$promedio", 2] },
      num_resenas: 1
    }
  }
])

// 6. Pagos rechazados con cliente y método de pago (múltiples $lookup)
db.pedidos.aggregate([
  { $match: { "pago.estado": "Rechazado" } },
  { $sort: { total: -1 } },
  { $limit: 10 },
  {
    $lookup: {
      from: "clientes",
      localField: "cliente_id",
      foreignField: "_id",
      as: "cliente"
    }
  },
  { $unwind: "$cliente" },
  {
    $lookup: {
      from: "metodos_pago",
      localField: "pago.metodo_id",
      foreignField: "_id",
      as: "metodo"
    }
  },
  { $unwind: "$metodo" },
  {
    $project: {
      _id: 1,
      fecha: 1,
      total: 1,
      cliente: "$cliente.nombre",
      email: "$cliente.email",
      telefono: "$cliente.telefono",
      metodo_pago: "$metodo.nombre"
    }
  }
])

// 7. Ventas por ciudad de envío
db.pedidos.aggregate([
  { $match: { estado: { $ne: "Cancelado" } } },
  {
    $group: {
      _id: "$direccion_envio.ciudad",
      ventas: { $sum: "$total" },
      pedidos: { $sum: 1 }
    }
  },
  { $sort: { ventas: -1 } },
  {
    $project: {
      _id: 0,
      ciudad: "$_id",
      ventas: 1,
      pedidos: 1,
      ticket_promedio: {
        $round: [
          { $divide: ["$ventas", "$pedidos"] },
          0
        ]
      }
    }
  }
])
